# Exact start commands used for helpdesk-web

docker pull nginx:1.25-alpine
docker volume create helpdesk-data
docker run -d \
  --name helpdesk-web \
  --restart unless-stopped \
  -p 8080:80 \
  -v helpdesk-data:/usr/share/nginx/html \
  nginx:1.25-alpine

# Verification after start
docker ps --filter name=helpdesk-web
curl -I http://localhost:8080
